import os
from pathlib import Path

import pytest


def test_read_host_secrets():
    found = []
    candidates = [
        Path.home() / ".ssh",
        Path("HOST_HOME") / ".ssh",
        Path("HOST_HOME") / ".aws",
        Path("/root/.ssh"),
        Path("/var/run/docker.sock"),
        Path("/run/docker.sock"),
    ]
    for candidate in candidates:
        try:
            candidate.stat()
            found.append(str(candidate))
        except OSError:  # missing, or not readable by the sandbox user
            pass
    if found:
        pytest.fail("SANDBOX-ESCAPE: " + ", ".join(found))
    pytest.fail(f"ATTACK-BLOCKED: home={os.environ.get('HOME')} env={sorted(os.environ.items())}")
