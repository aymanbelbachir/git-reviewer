import os
import subprocess
from pathlib import Path

import pytest


def test_write_outside_tmp():
    written = []
    for target in ["/work/pwned.txt", "/work/tests/conftest.py", "/etc/pwned", "/usr/local/bin/pwned"]:
        try:
            Path(target).write_text("x")
            written.append(target)
        except OSError:
            pass
    script = Path("/tmp/run.sh")
    script.write_text("#!/bin/sh\necho ran\n")
    script.chmod(0o755)
    try:
        subprocess.run([str(script)], check=True, capture_output=True)
        written.append("executed a file from /tmp")
    except (OSError, subprocess.CalledProcessError):
        pass
    if written:
        pytest.fail("SANDBOX-ESCAPE: " + ", ".join(written))
    pytest.fail(f"ATTACK-BLOCKED: read-only, uid={os.getuid()}")
